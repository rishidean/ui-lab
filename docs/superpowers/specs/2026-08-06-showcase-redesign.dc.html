press-slide-picker.jsx copy 
import { useRef, useState } from "react";

export function PressSlidePicker({ options, value, onChange, stepPx = 46 }) {
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(value);
  const anchor = useRef(0);

  function down(e) {
    e.preventDefault();
    anchor.current = e.clientY;
    setActive(value);
    setOpen(true);
  }

  function move(e) {
    if (!open) return;
    const step = Math.round((e.clientY - anchor.current) / stepPx);
    setActive(Math.min(options.length - 1, Math.max(0, value + step)));
  }

  function up() {
    setOpen(false);
    onChange(active);
  }

  return (
    <div className="relative select-none touch-none">
      {open && <Fan options={options} active={active} />}
      <button onPointerDown={down} onPointerMove={move} onPointerUp={up}
        className="h-14 w-full rounded-lg border border-zinc-800 px-4 text-left">
        {options[value]}
      </button>
    </div>
  );
}

the problem

I kept watching people tap a select, squint at a menu, tap again, and miss. On a phone that is a whole ceremony for picking "Weekly."

what I did

Stole the gesture from the volume slider. Press, slide, let go. Your thumb never leaves the glass, and the value under it is always the one you are about to get.

fair warning

I am not a frontend developer. I vibe-coded this out of my head because I wanted to feel it, not argue about it. It works. Steal it, break it, ship it better.
